(function(){
  const form=document.getElementById('bookingForm');
  const submitBtn=document.getElementById('submitBtn');
  const bookingWrap=document.getElementById('bookingWrap');
  const confirmBlock=document.getElementById('confirmBlock');
  const otherWrap=document.getElementById('otherEventWrap');
  const otherInput=document.getElementById('otherEvent');
  const state={event:null,time:null,guests:null,mood:null,atmosphere:null};

  document.querySelectorAll('.chips').forEach(group=>{
    const groupName=group.dataset.group;
    group.querySelectorAll('.chip').forEach(btn=>{
      btn.addEventListener('click',()=>{
        group.querySelectorAll('.chip').forEach(b=>b.classList.remove('on'));
        btn.classList.add('on');
        state[groupName]=btn.dataset.val;
        if(groupName==='event'){
          const isOther=btn.dataset.val==='Другое';
          otherWrap.classList.toggle('on',isOther);
          if(isOther)otherInput.focus();
        }
        updateSubmit();
      });
    });
  });

  function updateSubmit(){
    const basic=form.name.value.trim() && form.phone.value.trim() && form.date.value;
    const eventOk=state.event && (state.event!=='Другое' || otherInput.value.trim());
    const choices=eventOk && state.time && state.guests && state.mood && state.atmosphere;
    submitBtn.disabled=!(basic && choices);
  }

  document.querySelectorAll('.event-link').forEach(link=>{
    link.addEventListener('click',()=>{
      const chip=document.querySelector('#eventChips .chip[data-val="'+link.dataset.event+'"]');
      if(chip)chip.click();
      document.getElementById('booking').scrollIntoView({behavior:'smooth'});
    });
  });

  form.querySelectorAll('input').forEach(input=>input.addEventListener('input',updateSubmit));
  document.getElementById('date').min=new Date().toISOString().split('T')[0];

  form.addEventListener('submit',e=>{
    e.preventDefault();
    if(submitBtn.disabled)return;
    document.getElementById('confirmText').textContent='Ваша заявка: «'+(state.event==='Другое'?otherInput.value.trim():state.event)+'», гостей: '+state.guests+', время: '+state.time+'. Персональный менеджер свяжется с вами в течение 15 минут, чтобы обсудить детали мероприятия.';
    bookingWrap.style.display='none';
    confirmBlock.classList.add('on');
    confirmBlock.scrollIntoView({behavior:'smooth',block:'center'});
  });
})();
